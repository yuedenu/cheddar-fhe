cmake_minimum_required(VERSION 3.24)
project(cheddar_source LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CHEDDAR_CPU_ONLY "Build only the independent slot simulator" OFF)
option(CHEDDAR_LEGACY_REFERENCE "Build a separate historical-library driver" OFF)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(layout_test tests/layout_test.cpp)
target_include_directories(layout_test PRIVATE include)
add_test(NAME layout COMMAND layout_test)
add_executable(spatial_layout_test tests/spatial_layout_test.cpp)
target_include_directories(spatial_layout_test PRIVATE include)
add_test(NAME spatial_layout COMMAND spatial_layout_test)
add_test(NAME relu_basis COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/relu_basis_test.py)
if(CHEDDAR_CPU_ONLY)
  return()
endif()
if(NOT DEFINED CMAKE_CUDA_COMPILER AND EXISTS /usr/local/cuda-11.8/bin/nvcc)
  set(CMAKE_CUDA_COMPILER /usr/local/cuda-11.8/bin/nvcc)
endif()
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA architectures")
endif()
enable_language(CUDA)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(CUDAToolkit 11.8 REQUIRED)
find_library(TOMMATH_LIBRARY NAMES tommath REQUIRED)
find_package(ZLIB REQUIRED)
# All dependencies are source snapshots; see third_party/VERSIONS.json.
add_library(rmm_headers INTERFACE)
target_include_directories(rmm_headers SYSTEM INTERFACE
  third_party/rmm/include third_party/spdlog/include third_party/thrust
  third_party/thrust/dependencies/cub)
target_link_libraries(rmm_headers INTERFACE CUDA::cudart ${CMAKE_DL_LIBS})
file(GLOB CORE CONFIGURE_DEPENDS src/core/*.cpp src/core/*.cu)
list(FILTER CORE EXCLUDE REGEX "BigInt_gmp.cpp$")
file(GLOB EXTENSION CONFIGURE_DEPENDS src/extension/*.cpp src/extension/*.cu)
add_library(cheddar SHARED ${CORE} ${EXTENSION} src/UserInterface.cu)
set_target_properties(cheddar PROPERTIES OUTPUT_NAME cheddar_source)
target_include_directories(cheddar PUBLIC include)
target_compile_definitions(cheddar PUBLIC ENABLE_EXTENSION)
target_link_libraries(cheddar PUBLIC rmm_headers ${TOMMATH_LIBRARY})
add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
add_library(json_headers INTERFACE)
target_include_directories(json_headers SYSTEM INTERFACE third_party/jsoncpp/include)
foreach(target basic_test boot_test)
  if(target STREQUAL basic_test)
    set(test_source unittest/BasicTest.cpp)
  else()
    set(test_source unittest/Bootstrapping.cpp)
  endif()
  add_executable(${target} ${test_source})
  target_compile_definitions(${target} PRIVATE PARAM_DIR="${CMAKE_CURRENT_SOURCE_DIR}/unittest/parameters")
  target_link_libraries(${target} PRIVATE cheddar gtest_main json_headers)
endforeach()
add_library(cnpy_source STATIC third_party/cnpy/cnpy.cpp)
target_include_directories(cnpy_source PUBLIC third_party/cnpy)
target_link_libraries(cnpy_source PUBLIC ZLIB::ZLIB)
# The ResNet targets are declared in apps once the common driver is available.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/apps/CMakeLists.txt")
  add_subdirectory(apps)
endif()
add_executable(layer_error_analysis tests/layer_error_analysis.cpp)
target_include_directories(layer_error_analysis PRIVATE include)
target_link_libraries(layer_error_analysis PRIVATE cnpy_source json_headers)
