foreach(target resnet resnet_acc)
  add_executable(${target} main.cpp)
  target_link_libraries(${target} PRIVATE cheddar json_headers cnpy_source)
  target_compile_definitions(${target} PRIVATE CHEDDAR_ROOT="${PROJECT_SOURCE_DIR}" CHEDDAR_BACKEND="source")
  if(target STREQUAL resnet_acc)
    target_compile_definitions(${target} PRIVATE DEFAULT_COUNT=1000)
  else()
    target_compile_definitions(${target} PRIVATE DEFAULT_COUNT=1)
  endif()
endforeach()
